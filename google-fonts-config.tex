% google-fonts.tex — Compatible pdfLaTeX / XeLaTeX / LuaLaTeX

% 1) Détecter le moteur
\usepackage{iftex}

% 2) Couleurs et titres (utilisés plus bas)
\usepackage{xcolor}
\usepackage{titlesec}

% Définition des couleurs du thème
\definecolor{primaryBlue}{RGB}{33,81,156}      % Bleu principal professionnel
\definecolor{accentBlue}{RGB}{52,152,219}     % Bleu d'accent moderne
\definecolor{darkGray}{RGB}{51,51,51}         % Gris foncé pour le texte
\definecolor{lightGray}{RGB}{128,128,128}     % Gris clair pour les détails
\definecolor{backgroundGray}{RGB}{248,249,250} % Gris de fond très clair
\definecolor{successGreen}{RGB}{39,174,96}     % Vert pour les éléments positifs

% 3) Fonts selon le moteur
\ifPDFTeX
  % ----- pdfLaTeX : pas de Google Fonts -----
  \usepackage[T1]{fontenc}
  \usepackage[utf8]{inputenc}
  \usepackage{lmodern} % ou newtxtext,newtxmath si tu préfères
\else
  % ----- XeLaTeX / LuaLaTeX : Google Fonts via fontspec -----
  \usepackage{fontspec}
  \defaultfontfeatures{Ligatures=TeX, Scale=MatchLowercase}

  % --- Roboto en texte et sans-serif ---
  % Place les fichiers dans: fonts/Roboto/Roboto-*.ttf
  \setmainfont{Roboto}[
    Path=fonts/,
    Extension=.ttf,
    UprightFont   = *-Regular,
    BoldFont      = *-Bold,
    ItalicFont    = *-Italic,
    BoldItalicFont= *-BoldItalic
  ]
  \setsansfont{Roboto}[
    Path=fonts/,
    Extension=.ttf,
    UprightFont   = *-Regular,
    BoldFont      = *-Bold,
    ItalicFont    = *-Italic,
    BoldItalicFont= *-BoldItalic
  ]

  % --- (Optionnel) Roboto Mono pour le code ---
  % Fichiers dans: fonts/RobotoMono/RobotoMono-*.ttf
  % \setmonofont{RobotoMono}[
  %   Path=fonts/RobotoMono/,
  %   Extension=.ttf,
  %   UprightFont   = *-Regular,
  %   BoldFont      = *-Bold,
  %   ItalicFont    = *-Italic,
  %   BoldItalicFont= *-BoldItalic
  % ]
\fi

% Amélioration des sections avec le nouveau thème de couleurs
\titleformat{\section}
  {\fontsize{16pt}{19pt}\selectfont\bfseries\color{accentBlue}\sffamily}
  {\thesection}{1em}{}

\titleformat{\subsection}
  {\fontsize{14pt}{17pt}\selectfont\bfseries\color{primaryBlue}\sffamily}
  {\thesubsection}{1em}{}

\titleformat{\subsubsection}
  {\fontsize{12pt}{15pt}\selectfont\bfseries\color{darkGray}\sffamily}
  {\thesubsubsection}{1em}{}

% Amélioration des paragraphes et du texte général
\ifPDFTeX
  % Pour pdfLaTeX, améliorer l'espacement et la lisibilité
  \renewcommand{\baselinestretch}{1.15}
  \setlength{\parskip}{6pt plus 2pt minus 1pt}
  \setlength{\parindent}{1.5em}
  
  % Améliorer la typographie avec microtype
  \usepackage[activate={true,nocompatibility},final,tracking=true,kerning=true,spacing=true,factor=1100,stretch=10,shrink=10]{microtype}
  
  % Configuration pour de meilleures polices en pdfLaTeX
  \usepackage[scaled=0.92]{helvet} % Pour une meilleure police sans-serif
  % Garder la police par défaut (serif) pour le contenu
\else
  % Pour XeLaTeX/LuaLaTeX avec Google Fonts
  \renewcommand{\baselinestretch}{1.15}
  \setlength{\parskip}{6pt plus 2pt minus 1pt}
  \setlength{\parindent}{1.5em}
  
  % Améliorer l'affichage des citations et éléments spéciaux
  \newfontfamily\quotefont{Roboto}[
    Path=fonts/,
    Extension=.ttf,
    UprightFont=*-Italic
  ]
\fi

% Style pour les listes avec de meilleurs espacements
\usepackage{enumitem}
\setlist[itemize]{
  leftmargin=1.5em,
  itemsep=4pt plus 2pt minus 1pt,
  parsep=2pt plus 1pt minus 1pt,
  topsep=6pt plus 2pt minus 2pt
}

\setlist[enumerate]{
  leftmargin=1.8em,
  itemsep=4pt plus 2pt minus 1pt,
  parsep=2pt plus 1pt minus 1pt,
  topsep=6pt plus 2pt minus 2pt
}

% Amélioration des légendes de figures et tableaux avec le nouveau thème
\usepackage{caption}
\captionsetup{
  font={small,sf},
  labelfont={bf,color=primaryBlue},
  textfont={color=darkGray},
  margin=10pt,
  skip=8pt
}

% Modifier les intitulés des légendes pour les tableaux
\DeclareCaptionLabelFormat{tabformat}{Tableau~#2}
\captionsetup[table]{labelformat=tabformat}

% Style pour les références bibliographiques
\usepackage{url}
\renewcommand{\UrlFont}{\small\ttfamily}

% Amélioration de l'affichage des footnotes
\renewcommand{\footnotesize}{\fontsize{9pt}{11pt}\selectfont}

% Amélioration des en-têtes et pieds de page avec le nouveau thème
\usepackage{fancyhdr}
\fancypagestyle{thesis}{%
  \fancyhf{} % Clear all headers and footers
  \fancyhead[L]{\small\sffamily\color{darkGray}\leftmark}
  \fancyhead[R]{\small\sffamily\color{darkGray}\rightmark}
  \fancyfoot[C]{\sffamily\color{primaryBlue}\thepage}
  \renewcommand{\headrulewidth}{0.4pt}
  \renewcommand{\headrule}{\hbox to\headwidth{\color{accentBlue}\leaders\hrule height \headrulewidth\hfill}}
  \renewcommand{\footrulewidth}{0pt}
}

% Style pour les tableaux amélioré
\usepackage{booktabs}
\usepackage{array}
\renewcommand{\arraystretch}{1.2} % Espacement vertical dans les tableaux

% Amélioration des titres de niveau inférieur avec le nouveau thème
\titleformat{\paragraph}
  {\fontsize{11pt}{13pt}\selectfont\bfseries\color{accentBlue}\sffamily}
  {\theparagraph}{1em}{}

\titleformat{\subparagraph}
  {\fontsize{10pt}{12pt}\selectfont\bfseries\color{lightGray}\sffamily}
  {\thesubparagraph}{1em}{}

% Style pour les environnements de code avec le nouveau thème
\usepackage{listings}
\lstset{
    basicstyle=\footnotesize\ttfamily,
    commentstyle=\color{lightGray}\itshape,
    keywordstyle=\color{primaryBlue}\bfseries,
    stringstyle=\color{successGreen},
    numberstyle=\tiny\color{lightGray},
    showstringspaces=false,
    breaklines=true,
    frame=single,
    framerule=0.5pt,
    rulecolor=\color{accentBlue},
    backgroundcolor=\color{backgroundGray},
    tabsize=4,
    captionpos=b,
    xleftmargin=2em,
    xrightmargin=1em,
    framextopmargin=2pt,
    framexbottommargin=2pt
}

% Améliorations supplémentaires pour un thème moderne et attractif

% Amélioration des liens avec de meilleures couleurs
\usepackage{hyperref}
\hypersetup{
    colorlinks=true,
    linkcolor=primaryBlue,
    citecolor=accentBlue,
    urlcolor=accentBlue,
    filecolor=primaryBlue,
    menucolor=primaryBlue,
    anchorcolor=primaryBlue,
    pdfborder={0 0 0},
    breaklinks=true
}

% Amélioration de l'espacement autour des sections
\titlespacing*{\section}{0pt}{18pt plus 4pt minus 2pt}{12pt plus 2pt minus 2pt}
\titlespacing*{\subsection}{0pt}{16pt plus 3pt minus 2pt}{10pt plus 2pt minus 2pt}
\titlespacing*{\subsubsection}{0pt}{14pt plus 3pt minus 2pt}{8pt plus 2pt minus 2pt}

% Style moderne pour les footnotes
\renewcommand{\footnoterule}{%
  \kern-3pt
  {\color{accentBlue}\hrule width 0.4\columnwidth height 0.4pt}
  \kern 2.6pt
}

% Amélioration de l'affichage des références
\renewcommand{\bibname}{\color{primaryBlue}Bibliographie}

% Style pour les tables modernes
\usepackage{colortbl}
\arrayrulecolor{accentBlue}
\renewcommand{\arraystretch}{1.3}

\pagestyle{thesis}
